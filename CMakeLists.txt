cmake_minimum_required(VERSION 3.20)
project(kwin_relative_pointer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
include(KDECMakeSettings)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick DBus)
find_package(KF6CoreAddons REQUIRED)
find_package(KWin REQUIRED)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
kcoreaddons_add_plugin(kwin_relative_pointer SOURCES plugin.cpp INSTALL_NAMESPACE "kwin/effects/plugins")
target_link_libraries(kwin_relative_pointer PRIVATE KWin::kwin Qt6::Gui)
